182async def approve_waitlist_entry(entry_id: str, approved_by: str | None = None) -> dict | None:
183    """Promote a waitlist entry into the allowlist, unlocking sign-in.
184
185    Looks up the waitlist row, inserts a matching ``allowlist_entries`` row
186    via the existing allowlist service, marks the waitlist status as
187    ``approved``, then sends the approval email and records ``notified_at``
188    only when notification succeeds. The email is sent after the allowlist
189    write so the user is never notified before they are actually allowlisted.
190
191    Args:
192        entry_id: The waitlist entry id.
193        approved_by: Optional admin user id who approved the entry.
194
195    Returns:
196        dict | None: The resulting allowlist entry row.
197
198    Raises:
199        AuthError: When the waitlist entry does not exist or is not pending.
200    """
201    entry = await find_waitlist_entry_by_id(entry_id)
202    if not entry:
203        raise AuthError("Waitlist entry not found")
204    if entry["status"] != "pending":
205        raise AuthError(f"Waitlist entry cannot be approved (status: {entry['status']})")
206
207    now = datetime.now(UTC)
208    try:
209        async with get_transaction() as connection:
210            await add_entry(
211                entry_type=entry["entry_type"],
212                value=entry["value"],
213                note=entry.get("note"),
214                added_by=approved_by,
215                connection=connection,
216            )
217            await update_waitlist_entry(
218                entry_id=entry_id,
219                changes=WaitlistEntryUpdate(status="approved", approved_by=approved_by, approved_at=now),
220                connection=connection,
221            )
222    except AuthError:
223        # The allowlist row may already exist (e.g. an admin added it
224        # directly); promotion still succeeds in that case.
225        existing = await find_allowlist_entry(entry_type=entry["entry_type"], value=entry["value"])
226        if existing is None:
227            raise
228        await update_waitlist_entry(
229            entry_id=entry_id,
230            changes=WaitlistEntryUpdate(status="approved", approved_by=approved_by, approved_at=now),
231        )
232
233    try:
234        sent = send_approval_email(entry)
235    except (KeyError, OSError, RuntimeError):
236        logger.warning("Failed to send approval email for waitlist entry %s", entry_id, exc_info=True)
237        sent = False
238    if sent:
239        await update_waitlist_entry(
240            entry_id=entry_id,
241            changes=WaitlistEntryUpdate(status="approved", approved_by=approved_by, approved_at=now, notified_at=now),
242        )
243
244    return await find_allowlist_entry(entry_type=entry["entry_type"], value=entry["value"])

142@asynccontextmanager
143async def get_transaction(db_name: str | None = None) -> AsyncGenerator[AsyncSession]:
144    """Yield a session running inside an explicitly opened transaction.
145
146    The application engine runs at ``AUTOCOMMIT`` isolation, where
147    ``session.begin()`` is a no-op and every statement commits immediately.
148    This manager issues the transaction control itself (``BEGIN`` /
149    ``COMMIT`` / ``ROLLBACK``), so the yielded block runs atomically.
150
151    Args:
152        db_name: Optional database name; defaults to the configured DB_NAME.
153
154    Yields:
155        AsyncSession: A session inside an open transaction.
156    """
157    async with get_connection(db_name=db_name) as connection:
158        await connection.execute(text("BEGIN"))
159        try:
160            yield connection
161        except BaseException:
162            await connection.execute(text("ROLLBACK"))
163            raise
164        else:
165            await connection.execute(text("COMMIT"))
