27    """Send a GraphQL request to the Linear API and return its raw payload.
28
29    Resolves a valid OAuth token automatically.
30
31    Args:
32        query: The GraphQL query string.
33        variables: Optional query variables.
34
35    Returns:
36        dict: The raw JSON response payload.
37
38    Raises:
39        LinearError: When the request fails or the payload is unexpected.
40    """
41    token = await get_valid_token()
42
43    payload = {"query": query}
44    if variables is not None:
45        payload["variables"]: dict[str, Any] = variables  # ty: ignore
46
47    try:
48        async with aiohttp.ClientSession() as session:
49            async with session.post(
50                LINEAR["api_url"],
51                json=payload,
52                headers={
53                    "Authorization": f"Bearer {token}",
54                    "Content-Type": "application/json",
55                },
56                timeout=aiohttp.ClientTimeout(total=10),
57            ) as response:
58                response.raise_for_status()
59                data = await response.json()
60    except aiohttp.ClientError as e:
61        raise LinearError(f"Linear API error: {e}") from e
62
63    if not isinstance(data, dict):
64        raise LinearError(f"Linear API returned an unexpected payload: {data!r}")
65
66    return data
