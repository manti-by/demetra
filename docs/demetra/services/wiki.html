 70async def answer_sweep() -> int:
 71    """Apply answered ``## Open`` entries in QUESTIONS.md and move them to Resolved.
 72
 73    Returns:
 74        int: The number of questions resolved.
 75    """
 76    if not service.QUESTIONS_PATH.is_file():
 77        return 0
 78    text = await asyncio.to_thread(service.QUESTIONS_PATH.read_text, encoding="utf-8")
 79    if "## Open" not in text:
 80        return 0
 81
 82    header, _, rest = text.partition("## Open")
 83    open_block, _, resolved_block = rest.partition("## Resolved")
 84
 85    kept: list[str] = []
 86    resolved: list[str] = []
 87    current: list[str] = []
 88    preamble: list[str] = []
 89    in_entries = False
 90    count = 0
 91    for line in open_block.splitlines():
 92        if line.strip().startswith("### "):
 93            if current:
 94                entry_text = "\n".join(current)
 95                if service.has_answer(entry_text):
 96                    count += 1
 97                    resolved.append(entry_text)
 98                else:
 99                    kept.append(entry_text)
100            current = [line]
101            in_entries = True
102        elif in_entries:
103            current.append(line)
104        else:
105            preamble.append(line)
106    if current:
107        entry_text = "\n".join(current)
108        if service.has_answer(entry_text):
109            count += 1
110            resolved.append(entry_text)
111        else:
112            kept.append(entry_text)
113
114    if not count:
115        return 0
116
117    open_section = "\n\n".join(kept)
118    preamble_text = "\n".join(preamble).strip()
119    if open_section.strip() and preamble_text:
120        open_section = f"{preamble_text}\n\n{open_section}"
121    resolved_section = (
122        "\n\n".join([*resolved, resolved_block.strip()]) if resolved_block.strip() else "\n\n".join(resolved)
123    )
124    new_text = (
125        f"{header}## Open\n\n{open_section}\n\n## Resolved\n\n{resolved_section}\n"
126        if open_section.strip()
127        else f"{header}## Open\n\n_Newest first._\n\n## Resolved\n\n{resolved_section}\n"
128    )
129    await asyncio.to_thread(service.QUESTIONS_PATH.write_text, encoding="utf-8", data=new_text)
130    return count

334    """Assign a page to a ``## By topic`` cluster by keyword scoring.
335
336    Args:
337        meta: The page frontmatter mapping.
338
339    Returns:
340        str: The best-matching cluster title, or ``"Other"`` when none matches.
341    """
342    haystack = (
343        f"{meta.get('title') or ''} "
344        f"{' '.join(str(item) for item in (meta.get('services') or []))} "
345        f"{' '.join(str(item) for item in (meta.get('tags') or []))}"
346    ).casefold()
347    best = ("Other", 0)
348    for cluster, keywords in service.TOPIC_KEYWORDS.items():
349        score = sum(haystack.count(keyword) for keyword in keywords)
350        if score > best[1]:
351            best = (cluster, score)
352    return best[0]

380async def commit_revalidation(stats: dict) -> str | None:
381    """Commit the revalidation changes to the tracked files in ``stats``.
382
383    Only runs on the repository's default branch. Only the specific files the
384    revalidation sweep changed are staged and committed, never the whole
385    ``wiki/`` directory. When another process holds the git index lock,
386    ``REVALIDATION_RETRYABLE`` is returned so the caller can classify the
387    failure as transient rather than permanent.
388
389    Args:
390        stats: The revision statistics from ``revalidate_wiki_and_agents``.
391
392    Returns:
393        str | None: The commit SHA, ``REVALIDATION_RETRYABLE`` on index lock
394            contention, or None when there was nothing to commit.
395    """
396    changed = sorted(stats.get("changed_files") or [])
397    if not changed:
398        return None
399
400    if not await service.on_default_branch(target_path=service.BASE_PATH):
401        service.logger.warning(msg="Skipping wiki revalidation commit: BASE_PATH is not on the default branch")
402        return None
403
404    command = [str(service.GIT["path"]), "add", *changed]
405    exit_code, _, stderr = await service.run_command(command=command, target_path=service.BASE_PATH, disable_stdio=True)
406    if exit_code != 0:
407        if "index.lock" in stderr:
408            return service.REVALIDATION_RETRYABLE
409        service.logger.warning(msg=f"Failed to stage revalidation changes: {stderr.strip()}")
