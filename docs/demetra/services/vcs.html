309    """Clone a repository into the target path if it does not exist yet.
310
311    Args:
312        repo_url: The repository clone URL.
313        parent_path: Directory to run the GitHub CLI in.

317        dict: ``{"cloned": False}`` when the target already exists, otherwise

def extract_pr_link(stdout: str) -> str | None: 
47def extract_pr_link(stdout: str) -> str | None:
48    """Extract the PR URL from a command output line.
49
50    Args:
51        stdout: The command output to scan.
52
53    Returns:
54        str | None: The first GitHub pull request URL found, or None.
55    """
56    for line in stdout.splitlines():
57        candidate = line.strip().split(maxsplit=1)[0] if line.strip() else ""
58        parsed = urlparse(candidate)
59        if parsed.scheme not in {"http", "https"} or parsed.hostname not in GITHUB_HOSTS:
60            continue
61        if "pull" in parsed.path.split("/"):
62            return candidate
63    return None

452async def git_cleanup(context: Context, is_success: bool):
453    """Clean up a workflow's worktree and branch after a run finishes.
454
455    Always removes the worktree, forcing it on failure. On failure the branch
456    is also force-deleted. Research contexts run read-only in the main
457    checkout with no branch or worktree, so they are skipped entirely.
458
459    Args:
460        context: The workflow context describing the worktree and branch.
461        is_success: Whether the workflow completed successfully.
462    """
463    if context.is_research:
464        return
465
466    env = context.project.environment
467    project_id = context.project.id
468    try:
469        print_message("Removing worktree", style="heading")
470        await git_worktree_remove(
471            target_path=context.project.local_path,
472            worktree_path=context.worktree_path,
473            force=(not is_success),
474            env=env,
475            project_id=project_id,
476        )
477    except (OSError, RuntimeError, AttributeError):
478        print_message("Failed to remove worktree", style="error")
479
480    if is_success:
481        return
482
483    try:
484        print_message("Deleting branch", style="heading")
485        await git_branch_delete(
486            target_path=context.project.local_path, branch_name=context.branch_name, env=env, project_id=project_id
487        )
488    except (OSError, RuntimeError, AttributeError):
489        print_message("Failed to delete branch", style="error")

async def git_worktree_create( project: demetra.library.models.Project, branch_name: str, env: dict[str, str] | None = None, create_branch: bool = True) -> pathlib._local.Path: 
143async def git_worktree_create(
144    project: Project, branch_name: str, env: dict[str, str] | None = None, create_branch: bool = True
145) -> Path:
146    """Create a git worktree for a branch, cleaning up any stale worktree first.
147
148    When ``create_branch`` is False, the branch is force-created from the
149    remote tracking branch before the worktree is added. When True, an
150    existing local branch (e.g. a preserved orphan from a previous run) is
151    checked out instead of re-created, so its commits survive.
152
153    Args:
154        project: The project to operate on.
155        branch_name: The branch to check out in the worktree.
156        env: Optional environment overrides for the subprocess.
157        create_branch: Whether to create the branch with the worktree.
158
159    Returns:
160        Path: The path of the created worktree.
161
162    Raises:
163        RuntimeError: When branch or worktree creation fails.
164    """
165    worktree_path = get_worktree_path(project=project, branch_name=branch_name)
166    if worktree_path.exists():
167        git_file = worktree_path / ".git"
168        if git_file.exists() and git_file.is_file():
169            await git_worktree_remove(
170                target_path=project.local_path, worktree_path=worktree_path, force=True, env=env, project_id=project.id
171            )
172        else:
173            shutil.rmtree(worktree_path)
174        await git_delete_branch_preserving_unique(
175            target_path=project.local_path, branch_name=branch_name, env=env, project_id=project.id
176        )
177
178    worktree_path.parent.mkdir(parents=True, exist_ok=True)
179
180    if create_branch:
181        # If a previous run orphaned the branch (e.g. research success leaves the
182        # local branch), the worktree path no longer exists so the stale-branch
183        # cleanup above did not run. Delete it unless it holds unique commits.
184        if await git_branch_exists(target_path=project.local_path, branch_name=branch_name, env=env):
185            branch_preserved = await git_delete_branch_preserving_unique(
186                target_path=project.local_path, branch_name=branch_name, env=env, project_id=project.id
187            )
188        else:
189            branch_preserved = False
190        if branch_preserved:
191            command = [str(GIT["path"]), "worktree", "add", str(worktree_path), branch_name]
192        else:
193            command = [str(GIT["path"]), "worktree", "add", "-b", branch_name, str(worktree_path)]
194    else:
195        branch_cmd = [str(GIT["path"]), "branch", "--force", branch_name, f"origin/{branch_name}"]
196        branch_exit, _, branch_err = await run_command(
197            command=branch_cmd, target_path=project.local_path, env=env, project_id=project.id
198        )
199        if branch_exit != 0:
200            raise RuntimeError(f"Failed to create branch {branch_name}: {branch_err.strip() or 'unknown error'}")
201        command = [str(GIT["path"]), "worktree", "add", str(worktree_path), branch_name]
202
203    exit_code, _, stderr = await run_command(
204        command=command, target_path=project.local_path, env=env, project_id=project.id
205    )
206    if exit_code != 0:
207        raise RuntimeError(f"Failed to create worktree at {worktree_path}: {stderr.strip() or 'unknown error'}")

 50                    target_path=worktree_path,
 51                    env=env,
 52                    project_id=project_id,
 53                )
 54        else:
 55            logger.info(f"Successfully rebased branch {head_branch} onto {base_branch}")
 56        return True
 57
 58    logger.warning(f"Rebase with -X ours failed, attempting conflict resolution: {stderr.strip()[:500]}")
 59
